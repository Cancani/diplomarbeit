Direkte AWS-Nachkontrolle
Die folgenden Abfragen wurden nach delete ausgeführt. Sie verwenden keine
Agent-Tag-Filter.

ec2() {
  aws ec2 "$@" --profile diplomarbeit --region us-east-1 --output json --no-cli-pager
}

ec2 describe-instances \
  --filters 'Name=instance-state-name,Values=pending,running,stopping,stopped,shutting-down' \
  --query 'Reservations[].Instances[].{ID:InstanceId,Status:State.Name,Run:Tags[?Key==`RunId`].Value|[0]}'
ec2 describe-volumes --query 'Volumes[].{ID:VolumeId,Status:State,GiB:Size}'
ec2 describe-network-interfaces \
  --query 'NetworkInterfaces[].{ID:NetworkInterfaceId,Status:Status,VM:Attachment.InstanceId}'
ec2 describe-security-groups \
  --query 'SecurityGroups[?GroupName!=`default`].{ID:GroupId,Name:GroupName}'
ec2 describe-addresses --query 'Addresses[].{ID:AllocationId,VM:InstanceId}'
